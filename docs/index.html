sp_soft_delete_user

Parámetros IN/OUT: IN p_user_id INT UNSIGNED

Pasos: Valida la cuenta y la marca inactiva, preservando su historial.

Tablas afectadas: tb_users

CALL sp_soft_delete_user(@p_user_id);
Definición SQL
CREATE DEFINER=`root`@`localhost` PROCEDURE `sp_soft_delete_user`(
    IN p_user_id INT UNSIGNED
)
    MODIFIES SQL DATA
BEGIN
    DECLARE v_db_user VARCHAR(32);
    DECLARE v_exists INT DEFAULT 0;
    DECLARE v_status BIT DEFAULT NULL;
    DECLARE v_affected INT DEFAULT 0;

    -- 1. Identificar usuario de conexión
    SET v_db_user = SUBSTRING_INDEX(USER(), '@', 1);

    -- 2. Validar identificador
    IF p_user_id IS NULL OR p_user_id = 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'El identificador del usuario es inválido';
    END IF;

    -- 3. Comprobar existencia y estado actual del usuario
    SELECT COUNT(*), MAX(status)
    INTO v_exists, v_status
    FROM tb_users
    WHERE ID = p_user_id;

    IF v_exists = 0 OR v_status IS NULL THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'El usuario especificado no existe o tiene un estado no válido';
    END IF;

    -- 4. Comprobar si ya está inactivo (status = b'0')
    IF v_status = b'0' THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'El usuario ya se encuentra inactivo';
    END IF;

    -- 5. Eliminación lógica (Cambiar status a 0)
    UPDATE tb_users
    SET status = b'0'
    WHERE ID = p_user_id
      AND status = b'1';

    SET v_affected = ROW_COUNT();

    IF v_affected <> 1 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'No se pudo desactivar el usuario';
    END IF;

    -- 6. Mensaje de confirmación
    SELECT
        'Usuario desactivado correctamente' AS mensaje,
        p_user_id AS user_id,
        v_db_user AS ejecutado_por,
        'Inactivo (0)' AS estado_actual,
        CURRENT_TIMESTAMP AS fecha_operacion;
END ;;

Leyenda de la notación y criterios de clasificación

  • `=` significa “se compone de”; `+` significa “y”; `[a | b]` significa “uno de”; `{x}` significa repetición; `(x)` significa opcional; `*texto*` es comentario o dominio.
